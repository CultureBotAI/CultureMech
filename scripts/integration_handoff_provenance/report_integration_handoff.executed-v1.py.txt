#!/usr/bin/env python3
"""Audit a pinned CultureMech/MIM source handoff for issue #687 (2026-11-19).

Inputs: v3 bundle, archived MIM prepared/reviewed intake directories and three
independently reviewed manifest digests. Native producer/consumer verification
must be retained separately; this report does not replace either verifier.
Outputs: one exclusive JSON coverage report, including exact input/product pins,
case partitions and separate positive/context/held counts. No source is changed.
This is accounting, not a resolver, scientific approval, campaign adoption,
SSSOM certification, mapping installation or permission to publish a release.
"""

from __future__ import annotations

import argparse
import csv
import hashlib
import json
import re
import sys
from collections import Counter
from pathlib import Path


class HandoffError(ValueError):
    """The supplied accounting evidence is inconsistent."""


def fingerprint(path: Path) -> dict:
    digest = hashlib.sha256()
    size = 0
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
            size += len(chunk)
    return {"sha256": digest.hexdigest(), "bytes": size}


def pinned_json(path: Path, expected: str) -> dict:
    if not re.fullmatch(r"[0-9a-f]{64}", expected):
        raise HandoffError("Expected an independently supplied lowercase SHA256")
    raw = path.read_bytes()
    if hashlib.sha256(raw).hexdigest() != expected:
        raise HandoffError(f"Manifest pin mismatch: {path.name}")
    return json.loads(raw)


def checked_products(root: Path, products: dict) -> dict:
    result = {}
    for relative, expected in products.items():
        path = (root / relative).resolve()
        if not path.is_relative_to(root.resolve()):
            raise HandoffError("Product escapes its declared root")
        actual = fingerprint(path)
        if actual != {k: expected[k] for k in ("sha256", "bytes")}:
            raise HandoffError(f"Product mismatch: {relative}")
        result[relative] = actual
    return result


def rows(path: Path) -> list[dict]:
    previous = csv.field_size_limit()
    try:
        csv.field_size_limit(sys.maxsize)
        with path.open(encoding="utf-8", newline="") as stream:
            reader = csv.DictReader(stream, delimiter="\t")
            header = reader.fieldnames
            if not header or len(set(header)) != len(header):
                raise HandoffError(f"Missing or duplicate columns: {path.name}")
            result = list(reader)
            if any(None in row or None in row.values() for row in result):
                raise HandoffError(f"Ragged row: {path.name}")
            return result
    finally:
        csv.field_size_limit(previous)


def keyed(records: list[dict]) -> dict[str, dict]:
    result = {}
    for row in records:
        key = row["case_id"]
        if not key or key in result:
            raise HandoffError("Duplicate or empty binding case ID")
        result[key] = row
    return result


def summarize(records: list[dict]) -> dict:
    totals = Counter()
    actions = Counter()
    for row in records:
        for field in ("positive_occurrences", "context_only_mentions", "held_source_mentions"):
            value = row[field]
            if not re.fullmatch(r"0|[1-9][0-9]*", value):
                raise HandoffError(f"Invalid count: {field}")
            totals[field] += int(value)
        actions[row["change_action"] or "BASE_DIRECT"] += 1
    return {"binding_cases": len(records), **dict(totals), "cases_by_action": dict(actions)}


def audit(bundle: Path, intake: Path, reviewed: Path, pins: dict) -> dict:
    paths = {
        "source": bundle / "manifest.json",
        "intake": intake / "intake.json",
        "reviewed": reviewed / "reviewed-intake.json",
    }
    manifests = {key: pinned_json(path, pins[key]) for key, path in paths.items()}
    source, prepared, accepted = (manifests[key] for key in ("source", "intake", "reviewed"))
    if source.get("contract") != "culturemech-direct-ingredients-v3":
        raise HandoffError("Only the explicitly adopted v3 source contract is supported")
    if source.get("integrated_release") is not False:
        raise HandoffError("Source inventory cannot certify an integrated release")
    if accepted.get("status") != "REVIEWED_SOURCE_INTAKE":
        raise HandoffError("Reviewed native intake is required")
    for record in (prepared, accepted):
        if record.get("source_manifest_sha256") != pins["source"]:
            raise HandoffError("Source manifest relationship mismatch")
        if record.get("source_contract") != source["contract"]:
            raise HandoffError("Source contract mismatch")
    if accepted.get("intake_sha256") != pins["intake"]:
        raise HandoffError("Prepared intake relationship mismatch")
    if prepared.get("source_producer_commit") != source["git"]["commit"]:
        raise HandoffError("Producer revision mismatch")
    # Use the native exported products; never perform new identity resolution.
    source_products = checked_products(bundle, source["products"])
    reviewed_products = checked_products(reviewed, accepted["files"])
    names = ("binding-dispositions.tsv", "supported-bindings.tsv", "withheld-bindings.tsv")
    parsed = {name: rows(reviewed / name) for name in names}
    all_cases, supported, withheld = (keyed(parsed[name]) for name in names)
    if set(supported) & set(withheld) or set(all_cases) != set(supported) | set(withheld):
        raise HandoffError("Supported/withheld partitions must cover each case exactly once")
    for partition, verdict in ((supported, "SUPPORTED"), (withheld, "WITHHOLD")):
        for key, row in partition.items():
            if row != all_cases[key] or row["disposition"] != verdict:
                raise HandoffError("Partition differs from the full binding disposition")
    summaries = {
        "all": summarize(list(all_cases.values())),
        "supported": summarize(list(supported.values())),
        "withheld": summarize(list(withheld.values())),
    }
    counts = source["counts"]
    expected = {
        "positive_occurrences": counts["occurrences"],
        "context_only_mentions": counts["variant_context_only_mentions"],
        "held_source_mentions": counts["variant_held_mentions"],
    }
    if any(summaries["all"][key] != value for key, value in expected.items()):
        raise HandoffError("Binding coverage differs from positive/context/held source accounting")
    if len(all_cases) != prepared["counts"]["cases"]:
        raise HandoffError("Prepared case coverage mismatch")
    if accepted["counts"] != {"SUPPORTED": len(supported), "WITHHOLD": len(withheld)}:
        raise HandoffError("Reviewed case totals mismatch")
    if prepared["canonical_sssom"] != accepted["canonical_sssom"]:
        raise HandoffError("The separate intake must retain its unchanged canonical snapshot")
    # Read-back detects persistent mutations during this audit; no filesystem lock
    # or claim of safety against an adversary able to restore changed bytes.
    if checked_products(bundle, source["products"]) != source_products:
        raise HandoffError("Source products changed during audit")
    if checked_products(reviewed, accepted["files"]) != reviewed_products:
        raise HandoffError("Reviewed products changed during audit")
    for key, path in paths.items():
        pinned_json(path, pins[key])
    return {
        "schema_version": 1,
        "status": "SOURCE_BINDING_ACCOUNTING_VERIFIED",
        "integrated_release_ready": False,
        "producer_commit": source["git"]["commit"],
        "manifest_pins": pins,
        "source_counts": counts,
        "binding_coverage": summaries,
        "source_products": source_products,
        "reviewed_products": reviewed_products,
        "unchanged_historical_sssom": accepted["canonical_sssom"],
        "remaining_gates": [
            "Certify supported/withheld campaign SSSOM and required scoped companions",
            "Reconcile the selected KGM source revision and source deltas",
            "Independently admit exact release commit/archive/manifests in KGM",
            "Review paired mapping candidate and quarantine before installation",
            "Validate affected transforms, consumed-input receipts and graph outputs",
            "Obtain separate owner merge/publication approval",
        ],
        "limitations": [
            "Native archived producer and consumer verification remain separate requirements.",
            "Binding counts, occurrence counts and normalized ingredient groups are different units.",
            "No assertion of complete chemical-identity deduplication.",
            "Supported identity is not approval of dose, growth, or positive presence.",
            "Context and held counts remain separate even where identity has a disposition.",
            "Variant actions are not applied to parents or expanded into formulations.",
            "The existing canonical SSSOM snapshot is not the pending campaign release.",
            "Issue #792 NTA descriptor repair and unresolved source claims remain separate.",
        ],
    }


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ("bundle", "intake", "reviewed", "output"):
        parser.add_argument(f"--{name}", type=Path, required=True)
    for name in ("source", "intake", "reviewed"):
        parser.add_argument(f"--{name}-sha256", required=True)
    args = parser.parse_args(argv)
    try:
        result = audit(
            args.bundle,
            args.intake,
            args.reviewed,
            {key: getattr(args, f"{key}_sha256") for key in ("source", "intake", "reviewed")},
        )
        with args.output.open("x", encoding="utf-8") as stream:
            json.dump(result, stream, indent=2, sort_keys=True)
            stream.write("\n")
        print(json.dumps({"status": result["status"], "output": str(args.output)}))
    except (HandoffError, OSError, ValueError, KeyError, TypeError) as exc:
        print(f"Handoff accounting failed: {exc}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
